// --- FUNCIONES HARDWARE ---
void apagarTonoLimpiamente() {
  modoActual = 0;
  webDiagnostico = "En Espera...";
  ledcWriteTone(0, 0); 
  ledcDetachPin(txPins[3]); 
  for (int i = 0; i < 8; i++) {
    pinMode(txPins[i], OUTPUT);
    digitalWrite(txPins[i], LOW);
  }
}

// --- ENDPOINTS DEL SERVIDOR WEB ---
void manejarRaiz() { server.send(200, "text/html", paginaWeb); }

void manejarDatos() {
  String json = "{\"rx_map\":[";
  for (int i = 0; i < 8; i++) { json += String(webRxMap[i]); if (i < 7) json += ","; }
  json += "], \"calidad\":[";
  for (int i = 0; i < 8; i++) { json += String(webCalidad[i]); if (i < 7) json += ","; }
  json += "], \"modo\": " + String(modoActual) + ", \"paso_actual\": " + String(pasoSecuencia);
  json += ", \"diagnostico\": \"" + webDiagnostico + "\", \"consejo\": \"" + webConsejo + "\"}";
  
  server.send(200, "application/json", json);
}

void manejarComando() {
  String accion = server.arg("accion");
  
  if (accion == "apagar_tono" || accion == "apagar") {
    apagarTonoLimpiamente();
  } 
  else if (accion == "rapido") {
    apagarTonoLimpiamente(); modoActual = 1;
  } 
  else if (accion == "secuencial") {
    apagarTonoLimpiamente(); modoActual = 2; pasoSecuencia = 0; ultimoPaso = millis();
  } 
  else if (accion == "tono") {
    apagarTonoLimpiamente();
    modoActual = 3;
    tiempoInicioTono = millis();
    ledcAttachPin(txPins[3], 0); 
    webDiagnostico = "Emitiendo Tono Sirena...";
  }
  server.send(200, "text/plain", "Comando OK");
}

void manejarReporte() {
  String csv = "Pin TX,Color,Pin RX Destino,Estado,Calidad Ponchado (%)\n";
  for (int i = 0; i < 8; i++) {
    String estado = "OK";
    if (webRxMap[i] == -1) estado = "ROTO";
    else if (webRxMap[i] != i) estado = "CRUZADO";
    csv += String(i+1) + "," + colores[i] + "," + String(webRxMap[i]+1) + "," + estado + "," + String(webCalidad[i]) + "%\n";
  }
  server.sendHeader("Content-Disposition", "attachment; filename=\"Certificacion_MexWeb.csv\"");
  server.send(200, "text/csv", csv);
}

void setup() {
  Serial.begin(115200);
  ledcSetup(0, 1000, 8); 

  for (int i = 0; i < 8; i++) {
    pinMode(txPins[i], OUTPUT); digitalWrite(txPins[i], LOW);
    pinMode(rxPins[i], INPUT_PULLDOWN); 
  }
  
  if(!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) { for(;;); }
  
  display.clearDisplay(); display.setTextSize(1); display.setTextColor(WHITE);
  display.setCursor(7, 10); display.println("Tester MexWeb V1.06 AP");
  display.setCursor(0, 30); display.println("Creando Red WiFi...");
  display.display();
  
  // CONFIGURACIÓN DE MODO PUNTO DE ACCESO (AP)
  WiFi.softAPConfig(local_IP, gateway, subnet);
  WiFi.softAP(ssid, password);
  
  IPAddress myIP = WiFi.softAPIP();

  display.clearDisplay(); display.setCursor(7, 10); display.println("Tester MexWeb V1.06 AP");
  display.setCursor(0, 25); display.println("Red Creada OK!"); 
  display.setCursor(0, 40); display.print("SSID: "); display.println(ssid);
  display.setCursor(0, 50); display.print("IP:   "); display.println(myIP);
  
  server.on("/", manejarRaiz);
  server.on("/datos", manejarDatos);
  server.on("/cmd", manejarComando);
  server.on("/reporte", manejarReporte);
  server.begin();
  
  display.display();
  delay(4000); 
}

void loop() {
  server.handleClient(); 
  unsigned long tiempoActual = millis();

  // --- MODO 0: ESPERA ---
  if (modoActual == 0) {
    webConsejo = "Selecciona un modo de prueba en los botones de abajo.";
    display.clearDisplay(); display.setTextSize(1); display.setTextColor(WHITE);
    display.setCursor(1, 0); display.println("TESTER CABLE UTP"); display.drawLine(0, 10, 128, 10, WHITE); 
    display.setCursor(0, 30); display.println("   [ EN ESPERA ]"); display.setCursor(0, 50); display.println(" Seleccione prueba");
    display.display(); delay(100); return;
  }

  // --- MODO 3: TONO SIRENA ---
  if (modoActual == 3) {
    webConsejo = "ESTATUS: Sirena activa. Rastrea el cable con el pollo. Pulsa PARO para terminar.";
    unsigned long tiempoSonando = millis() - tiempoInicioTono;
    if (tiempoSonando > 30000) { apagarTonoLimpiamente(); } 
    else {
      if ((tiempoSonando / 250) % 2 == 0) ledcWriteTone(0, 800); else ledcWriteTone(0, 1000);
      display.clearDisplay(); display.setCursor(1, 0); display.println("TESTER CABLE UTP"); display.drawLine(0, 10, 128, 10, WHITE); 
      display.setCursor(15, 30); display.println("EMITIENDO TONO"); display.setCursor(20, 45); display.println("Hilo Azul (Sirena)");
      display.display(); delay(50);
    }
    return;
  }

  // --- MODO 1 y 2: ESCANEO Y CALIDAD ---
  int numMalos = 0; bool hayCruce = false;
  bool estadoParpadeo = ((tiempoActual / 250) % 2) == 0;

  if (modoActual == 1) { // RÁPIDO
    for (int i = 0; i < 8; i++) {
      digitalWrite(txPins[i], HIGH);
      unsigned long t1 = micros();
      
      webRxMap[i] = -1; 
      while(micros() - t1 < 1000) {
        for (int j = 0; j < 8; j++) { 
          if (digitalRead(rxPins[j]) == HIGH) { webRxMap[i] = j; break; }
        }
        if(webRxMap[i] != -1) break;
      }
      unsigned long tiempoViaje = micros() - t1;
      digitalWrite(txPins[i], LOW);
      
      if (webRxMap[i] == -1) webCalidad[i] = 0;
      else if (tiempoViaje < 25) webCalidad[i] = 100; 
      else if (tiempoViaje < 100) webCalidad[i] = 90; 
      else if (tiempoViaje < 500) webCalidad[i] = 70; 
      else webCalidad[i] = 50; 

      if (webRxMap[i] == -1) numMalos++; else if (webRxMap[i] != i) hayCruce = true;
    }
  } 
  else if (modoActual == 2) { // SECUENCIAL
    if (tiempoActual - ultimoPaso > 800) { 
      pasoSecuencia++; if (pasoSecuencia > 7) pasoSecuencia = 0; ultimoPaso = tiempoActual;
    }
    
    digitalWrite(txPins[pasoSecuencia], HIGH);
    unsigned long t1 = micros();
    webRxMap[pasoSecuencia] = -1;
    
    while(micros() - t1 < 1000) {
      for (int j = 0; j < 8; j++) { if (digitalRead(rxPins[j]) == HIGH) { webRxMap[pasoSecuencia] = j; break; } }
      if(webRxMap[pasoSecuencia] != -1) break;
    }
    unsigned long tiempoViaje = micros() - t1;
    digitalWrite(txPins[pasoSecuencia], LOW);

    if (webRxMap[pasoSecuencia] == -1) webCalidad[pasoSecuencia] = 0;
    else if (tiempoViaje < 25) webCalidad[pasoSecuencia] = 100;
    else if (tiempoViaje < 100) webCalidad[pasoSecuencia] = 90;
    else if (tiempoViaje < 500) webCalidad[pasoSecuencia] = 70;
    else webCalidad[pasoSecuencia] = 50;

    for (int i = 0; i < 8; i++) { if (i != pasoSecuencia) webRxMap[i] = -2; } 
    if (webRxMap[pasoSecuencia] == -1) numMalos++; else if (webRxMap[pasoSecuencia] != pasoSecuencia) hayCruce = true;
  }

  // --- GENERADOR DEL CONSEJO/ACCIÓN PARA LA WEB ---
  webConsejo = "";
  if (numMalos == 8 && !hayCruce) {
    webConsejo = "Conecta el cable en ambos puertos (TX y RX) para iniciar el diagnóstico.";
  } else if (numMalos == 0 && !hayCruce) {
    webConsejo = "ESTATUS: Transmitiendo 100%. Cable en óptimas condiciones.";
  } else {
    webConsejo = "ACCIÓN: REVISAR ";
    bool primero = true;
    for (int i=0; i<8; i++) {
      if (webRxMap[i] == -1 || (webRxMap[i] != i && webRxMap[i] != -2)) {
        if (!primero) webConsejo += ", ";
        webConsejo += colores[i];
        primero = false;
      }
    }
    webConsejo += ".";
  }

  // --- DIBUJAR PANTALLA OLED ---
  display.clearDisplay(); display.setTextSize(1); display.setTextColor(WHITE, BLACK); 
  display.setCursor(1, 0); display.println("TESTER CABLE UTP"); display.drawLine(0, 10, 128, 10, WHITE); 

  display.setCursor(0, 22); display.print("TX: ");
  for (int i = 0; i < 8; i++) {
    if ((modoActual == 1 && webRxMap[i] == -1) || (modoActual == 2 && webRxMap[pasoSecuencia] == -1 && i == pasoSecuencia)) {
      if(estadoParpadeo) display.print("X "); else display.print("  ");
    } else {
      if (modoActual == 2 && i == pasoSecuencia) { display.setTextColor(BLACK, WHITE); display.print(i + 1); display.setTextColor(WHITE, BLACK); } 
      else { display.print(i + 1); }
      display.print(" ");
    }
  }

  display.setCursor(0, 36); display.print("RX: ");
  for (int j = 0; j < 8; j++) {
    bool rxRecibe = false;
    for(int k=0; k<8; k++) { if(webRxMap[k] == j) rxRecibe = true; }

    if ((modoActual == 1 && !rxRecibe) || (modoActual == 2 && webRxMap[pasoSecuencia] == -1 && j == pasoSecuencia)) {
      if(estadoParpadeo) display.print("X "); else display.print("  ");
    } else {
      if (modoActual == 2 && webRxMap[pasoSecuencia] == j) { display.setTextColor(BLACK, WHITE); display.print(j + 1); display.setTextColor(WHITE, BLACK); } 
      else { display.print(j + 1); }
      display.print(" ");
    }
  }

  display.setCursor(0, 54);
  if (modoActual == 1) { 
    if (numMalos == 8) { display.println("Sin cable conectado"); webDiagnostico = "Desconectado"; }
    else if (numMalos > 0) { display.println("Falla: Cable Roto"); webDiagnostico = "Falla: Cable Roto"; }
    else if (hayCruce) { display.println("Falla: Cruzado"); webDiagnostico = "Falla: Cruzado"; }
    else { display.println("Secuencia Correcta"); webDiagnostico = "100% OK"; }
  } else { 
    display.print("Checando: "); display.println(colores[pasoSecuencia]);
    webDiagnostico = "Secuencia Activa";
  }

  display.display(); delay(15); 
}
